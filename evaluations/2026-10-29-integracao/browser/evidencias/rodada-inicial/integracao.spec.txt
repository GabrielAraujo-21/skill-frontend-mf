import { test, expect } from '@playwright/test';
import AxeBuilder from '@axe-core/playwright';

async function entrar(page, email = 'a@example.test') {
  await page.goto('/login');
  await page.getByLabel('E-mail', { exact: true }).fill(email);
  await page.getByLabel('Senha', { exact: true }).fill('Teste123456!');
  await page.getByRole('button', { name: 'Entrar', exact: true }).click();
  await expect(page).toHaveURL(/\/clientes$/);
  await expect(page.getByTestId('identidade')).toBeVisible();
}

async function requisitar(page, caminho, metodo = 'GET', dados) {
  return page.evaluate(async ({ caminho, metodo, dados }) => {
    const csrf = document.cookie.split('; ').find((parte) => parte.startsWith('XSRF-TOKEN='));
    const resposta = await fetch(caminho, {
      method: metodo,
      headers: { Accept: 'application/json', 'Content-Type': 'application/json',
        ...(csrf ? { 'X-XSRF-TOKEN': decodeURIComponent(csrf.slice('XSRF-TOKEN='.length)) } : {}) },
      ...(dados ? { body: JSON.stringify(dados) } : {}),
    });
    return { status: resposta.status, corpo: await resposta.text() };
  }, { caminho, metodo, dados });
}

test('rota protegida, credencial incorreta e CSRF real', async ({ page }) => {
  await page.goto('/clientes');
  await expect(page).toHaveURL(/\/login$/);
  await page.getByLabel('E-mail', { exact: true }).fill('a@example.test');
  await page.getByLabel('Senha', { exact: true }).fill('errada');
  await page.getByRole('button', { name: 'Entrar', exact: true }).click();
  await expect(page).toHaveURL(/\/login$/);
  const status = await page.evaluate(async () => (await fetch('/api/login', {
    method: 'POST', headers: { Accept: 'application/json', 'Content-Type': 'application/json' },
    body: JSON.stringify({ email: 'a@example.test', password: 'Teste123456!' }),
  })).status);
  expect(status).toBe(419);
  expect((await requisitar(page, '/api/clientes')).status).toBe(401);
});

test('cadastro CNPJ alfanumérico, edição e isolamento entre organizações', async ({ page }) => {
  await entrar(page);
  const nome = `Cliente navegador ${Date.now()}`;
  await page.getByLabel('Nome', { exact: true }).fill(nome);
  await page.getByLabel('CNPJ', { exact: true }).fill('12.ABC.345/01DE-35');
  const criado = page.waitForResponse((r) => r.url().endsWith('/api/clientes') && r.request().method() === 'POST');
  await page.getByRole('button', { name: 'Salvar cliente', exact: true }).click();
  const resposta = await criado;
  expect(resposta.status()).toBe(201);
  const { data: cliente } = await resposta.json();
  await expect(page.getByTestId('clientes')).toContainText(nome);
  await page.getByRole('button', { name: `Editar ${nome}`, exact: true }).click();
  await page.getByLabel('Nome', { exact: true }).fill(`${nome} atualizado`);
  await page.getByRole('button', { name: 'Salvar cliente', exact: true }).click();
  await expect(page.getByTestId('clientes')).toContainText(`${nome} atualizado`);
  await page.getByRole('button', { name: 'Sair', exact: true }).click();
  await entrar(page, 'b@example.test');
  await expect(page.getByTestId('clientes')).not.toContainText(nome);
  const tentativa = await requisitar(page, `/api/clientes/${cliente.id}`, 'PUT', { nome: 'Ataque sintético', cnpj: cliente.cnpj });
  expect([403, 404]).toContain(tentativa.status);
});

test('validação servidor permanece ativa quando cliente é contornado', async ({ page }) => {
  await entrar(page);
  const resposta = await requisitar(page, '/api/clientes', 'POST', { nome: '', cnpj: '12ABC34501DE34' });
  expect(resposta.status).toBe(422);
  const dados = JSON.parse(resposta.corpo);
  expect(dados.errors).toHaveProperty('nome');
  expect(dados.errors).toHaveProperty('cnpj');
});

test('logout real revoga sessão e impede retorno após recarga', async ({ page }) => {
  await entrar(page);
  await page.getByRole('button', { name: 'Sair', exact: true }).click();
  await expect(page).toHaveURL(/\/login$/);
  expect((await requisitar(page, '/api/me')).status).toBe(401);
  await page.reload();
  await expect(page).toHaveURL(/\/login$/);
  await expect(page.getByTestId('clientes')).toHaveCount(0);
});

test('logout offline não restaura dados após recarga e permite reconciliar', async ({ page }) => {
  await entrar(page);
  await page.route('**/api/logout', (rota) => rota.abort('failed'));
  await page.getByRole('button', { name: 'Sair', exact: true }).click();
  await expect(page).toHaveURL(/\/login$/);
  await expect(page.getByTestId('logout-pendente')).toBeVisible();
  await page.reload();
  await expect(page.getByTestId('logout-pendente')).toBeVisible();
  await expect(page.getByTestId('clientes')).toHaveCount(0);
  await page.unroute('**/api/logout');
  await page.getByRole('button', { name: /Tentar encerrar|Reconciliar|Tentar novamente/i }).click();
  await expect(page.getByTestId('logout-pendente')).toHaveCount(0);
  expect((await requisitar(page, '/api/me')).status).toBe(401);
});

test('resposta antiga após logout e troca de conta não repovoa a tela', async ({ page }) => {
  await entrar(page);
  await expect(page.getByRole('button', { name: 'Atualizar lista' })).toBeEnabled();
  let liberar;
  let recebida;
  let nomesAntigos = [];
  const chegou = new Promise((resolve) => { recebida = resolve; });
  const aguardar = new Promise((resolve) => { liberar = resolve; });
  await page.route('**/api/clientes', async (rota) => {
    const resposta = await rota.fetch();
    nomesAntigos = (await resposta.json()).data.map((cliente) => cliente.nome);
    recebida();
    await aguardar;
    await rota.fulfill({ response: resposta });
  }, { times: 1 });
  await page.getByRole('button', { name: 'Atualizar lista' }).click();
  await chegou;
  await page.getByRole('button', { name: 'Sair', exact: true }).click();
  await expect(page).toHaveURL(/\/login$/);
  await expect(page.getByTestId('clientes')).toHaveCount(0);
  await entrar(page, 'b@example.test');
  await expect(page.getByTestId('identidade')).toBeVisible();
  const terminou = page.waitForEvent('requestfinished', { predicate: (req) => req.url().endsWith('/api/clientes') });
  liberar();
  await terminou;
  // Deixar as microtasks do handler Axios e o próximo frame Vue concluírem.
  await page.evaluate(() => new Promise((resolve) => requestAnimationFrame(() => requestAnimationFrame(resolve))));
  for (const nome of nomesAntigos) await expect(page.getByTestId('clientes')).not.toContainText(nome);
});

test('logout em outra aba limpa dados e reconcilia sessão já revogada', async ({ page, context }) => {
  await entrar(page);
  const outra = await context.newPage();
  await outra.goto('/clientes');
  await expect(outra.getByTestId('identidade')).toBeVisible();
  await page.getByRole('button', { name: 'Sair', exact: true }).click();
  await expect(page).toHaveURL(/\/login$/);
  await expect(page.getByTestId('logout-pendente')).toHaveCount(0);
  await expect(outra).toHaveURL(/\/login$/);
  await expect(outra.getByTestId('clientes')).toHaveCount(0);
  await outra.getByRole('button', { name: /Tentar encerrar novamente/i }).click();
  await expect(outra.getByTestId('logout-pendente')).toHaveCount(0);
});

test('acessibilidade automatizada, teclado e viewport móvel', async ({ page }) => {
  await page.setViewportSize({ width: 390, height: 844 });
  await page.goto('/login');
  await page.getByLabel('E-mail', { exact: true }).focus();
  await page.keyboard.type('a@example.test');
  await page.keyboard.press('Tab');
  await expect(page.getByLabel('Senha', { exact: true })).toBeFocused();
  await page.keyboard.type('Teste123456!');
  await page.keyboard.press('Enter');
  await expect(page).toHaveURL(/\/clientes$/);
  await expect(page.getByTestId('clientes')).toBeVisible();
  const analise = await new AxeBuilder({ page }).withTags(['wcag2a', 'wcag2aa', 'wcag21aa']).analyze();
  expect(analise.violations).toEqual([]);
  expect(await page.evaluate(() => document.documentElement.scrollWidth <= window.innerWidth)).toBeTruthy();
  await page.screenshot({ path: 'evidencias/clientes-mobile.png', fullPage: true });
});
